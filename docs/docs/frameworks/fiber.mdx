# Fiber

[Fiber](https://gofiber.io) runs on fasthttp rather than `net/http`, so goauth's handlers are connected with Fiber's [`adaptor`](https://docs.gofiber.io/next/middleware/adaptor) package.

:::info[Reading the caller in native Fiber handlers]
A `net/http` middleware stores the validated token on the `*http.Request` context, and that context does not reach the next Fiber handler. The `withAuth` helper below solves this: it runs the goauth middleware, and when the request is allowed it saves the caller (`goauth.AuthData`) in `c.Locals("auth")`.
:::

## Install

```bash
go get github.com/izetmolla/goauth github.com/gofiber/fiber/v3 github.com/glebarez/sqlite gorm.io/gorm github.com/google/uuid
```

## Adapters at a glance

| goauth value | Fiber adapter |
|---|---|
| `auth.Handler()` | `app.All(goauth.DefaultBasePath+"/*", adaptor.HTTPHandler(auth.Handler()))` |
| an `http.HandlerFunc` | `adaptor.HTTPHandlerFunc(f)` |
| a goauth middleware (pure net/http chain) | `adaptor.HTTPMiddleware(mw)` |
| a goauth middleware + native Fiber handler | `withAuth(mw, extractor)` (below) |

```go title="setup.go"
package main

// setup.go — shared by every framework example on these pages.
// It opens the database, configures goauth and defines password
// sign-in and sign-out as plain net/http handlers.

import (
	"cmp"
	"context"
	"encoding/json"
	"errors"
	"io"
	"log"
	"net/http"
	"os"
	"strings"

	"github.com/glebarez/sqlite"
	"github.com/google/uuid"
	"github.com/izetmolla/goauth"
	"github.com/izetmolla/goauth/providers/google"
	"gorm.io/gorm"
)

func newAuth() (*goauth.Authorization, *gorm.DB) {
	db, err := gorm.Open(sqlite.Open("app.db"), &gorm.Config{})
	if err != nil {
		log.Fatal(err)
	}
	if err := migrate(db); err != nil {
		log.Fatal(err)
	}
	auth, err := goauth.New(&goauth.Config{
		JWTSecret:           cmp.Or(os.Getenv("JWT_SECRET"), "dev-secret-change-me"),
		AuthURL:             "http://localhost:3000",
		AccessTokenDuration: "15m",
		DB:                  db,
		Providers: []goauth.Provider{
			google.New(os.Getenv("GOOGLE_CLIENT_ID"), os.Getenv("GOOGLE_CLIENT_SECRET")),
		},
		ResolveUser: func(ctx context.Context, p goauth.Profile) (*goauth.User, bool, error) {
			email, _ := p["email"].(string)
			return findOrCreateUser(ctx, db, email)
		},
	})
	if err != nil {
		log.Fatal(err)
	}
	return auth, db
}

// findOrCreateUser returns the user with this email, creating it on first sign-in.
func findOrCreateUser(ctx context.Context, db *gorm.DB, email string) (*goauth.User, bool, error) {
	email = strings.ToLower(strings.TrimSpace(email))
	if email == "" {
		return nil, false, errors.New("email is required")
	}
	var u goauth.User
	err := db.WithContext(ctx).Table("users").Select("id", "roles").Where("email = ?", email).Take(&u).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		u = goauth.User{ID: uuid.NewString(), Roles: goauth.JSONBArray{"user:r"}}
		err = db.WithContext(ctx).Exec(`INSERT INTO users (id, email, roles) VALUES (?, ?, ?)`,
			u.ID, email, u.Roles).Error
		return &u, true, err
	}
	return &u, false, err
}

// passwordLogin signs a user in with email + password and returns the token pair.
func passwordLogin(auth *goauth.Authorization, db *gorm.DB) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		var in struct{ Email, Password string }
		if err := json.NewDecoder(io.LimitReader(r.Body, 1<<20)).Decode(&in); err != nil {
			http.Error(w, "invalid JSON", http.StatusBadRequest)
			return
		}
		var u struct{ ID, PasswordHash string }
		err := db.WithContext(r.Context()).Table("users").Select("id", "password_hash").
			Where("email = ?", strings.ToLower(in.Email)).Take(&u).Error
		if err != nil || !goauth.CheckPassword(u.PasswordHash, in.Password) {
			http.Error(w, "invalid email or password", http.StatusUnauthorized)
			return
		}
		opts := append([]goauth.AuthorizeOptionsFunc{auth.WithUserID(u.ID)},
			auth.SessionMetaFromRequest(r, "password")...)
		tokens, sessionID, err := auth.Authorize(opts...)
		if err != nil {
			http.Error(w, "could not create session", http.StatusInternalServerError)
			return
		}
		auth.SetSessionIDCookie(w, r, sessionID)
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(map[string]any{"session_id": sessionID, "user_id": u.ID, "tokens": tokens})
	}
}

// logout ends the session found in the cookie and expires the cookie.
func logout(auth *goauth.Authorization, db *gorm.DB) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		if id := auth.GetSessionID(r); id != "" {
			db.WithContext(r.Context()).Table("sessions").Where("id = ?", id).Update("is_deleted", true)
		}
		http.SetCookie(w, &http.Cookie{Name: goauth.DefaultCookieSessionName, Path: "/", MaxAge: -1, HttpOnly: true})
		w.WriteHeader(http.StatusNoContent)
	}
}

// refreshFallback answers refresh requests that lack the "cft" opt-in header.
var refreshFallback = http.HandlerFunc(func(w http.ResponseWriter, _ *http.Request) {
	http.Error(w, `send the "cft: 1" header to refresh a token`, http.StatusBadRequest)
})

func migrate(db *gorm.DB) error {
	return db.Exec(`
CREATE TABLE IF NOT EXISTS users (
	id            TEXT PRIMARY KEY,
	email         TEXT UNIQUE,
	password_hash TEXT,
	roles         TEXT DEFAULT '[]',
	content       TEXT DEFAULT '{}'
);
CREATE TABLE IF NOT EXISTS sessions (
	id TEXT PRIMARY KEY, user_id TEXT, type TEXT DEFAULT 'sign_in',
	ip_address TEXT, user_agent TEXT, method TEXT DEFAULT 'credentials',
	account TEXT, expires_at DATETIME, is_deleted NUMERIC DEFAULT 0,
	created_at DATETIME, updated_at DATETIME, deleted_at DATETIME
);`).Error
}
```

## The app

```go title="main.go"
package main

import (
	"log"
	"net/http"

	"github.com/gofiber/fiber/v3"
	"github.com/gofiber/fiber/v3/middleware/adaptor"
	"github.com/izetmolla/goauth"
)

func main() {
	auth, db := newAuth()
	app := fiber.New()

	// 1. goauth endpoints.
	app.All(goauth.DefaultBasePath+"/*", adaptor.HTTPHandler(auth.Handler()))

	// 2. Password sign-in, token refresh and sign-out (net/http handlers).
	app.Post("/api/login", adaptor.HTTPHandlerFunc(passwordLogin(auth, db)))
	app.Post("/api/token/refresh", adaptor.HTTPHandler(auth.HandleRefreshToken(refreshFallback)))
	app.Post("/api/logout", adaptor.HTTPHandlerFunc(logout(auth, db)))

	// 3. Build Fiber middlewares from goauth middlewares once.
	requireJWT := withAuth(auth.UseAPIAuthorization(), auth.GetAuthDataAPI)
	requireAdmin := withAuth(auth.UseAPIAuthorization(auth.WithRoles([]string{"admin"})), auth.GetAuthDataAPI)
	requireSession := withAuth(auth.UseWEBAuthorization(), func(r *http.Request) (goauth.AuthData, error) {
		return auth.GetAuthDataWEB(r, r.Context())
	})

	// 4. Native Fiber handlers read the caller from c.Locals.
	app.Get("/api/me", requireJWT, func(c fiber.Ctx) error {
		me := fiber.Locals[goauth.AuthData](c, "auth")
		return c.JSON(fiber.Map{"user_id": me.UserID, "roles": me.Roles})
	})

	// 5. Route groups: every route in /api/admin requires the admin role.
	admin := app.Group("/api/admin", requireAdmin)
	admin.Get("/stats", func(c fiber.Ctx) error {
		return c.JSON(fiber.Map{"users": 42})
	})

	// 6. A server-rendered page protected by the session cookie.
	app.Get("/dashboard", requireSession, func(c fiber.Ctx) error {
		me := fiber.Locals[goauth.AuthData](c, "auth")
		return c.SendString("Hello " + me.UserID)
	})

	app.Get("/sign-in", func(c fiber.Ctx) error {
		c.Type("html")
		return c.SendString(`<a href="/api/authorization/provider/google">Sign in with Google</a>`)
	})

	log.Fatal(app.Listen(":3000"))
}

// withAuth turns a goauth middleware into a Fiber handler. When the request
// is allowed, the caller is stored in c.Locals("auth") for native handlers;
// otherwise goauth's 401/403 response or sign-in redirect is sent as is.
//
// It is needed because a net/http middleware's request context does not reach
// the next Fiber handler, so goauth.GetClaims cannot be used there directly.
func withAuth(mw goauth.Middleware, caller func(*http.Request) (goauth.AuthData, error)) fiber.Handler {
	return func(c fiber.Ctx) error {
		var data *goauth.AuthData
		check := mw(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			if d, err := caller(r); err == nil {
				data = &d
			} else {
				http.Error(w, err.Error(), http.StatusUnauthorized)
			}
		}))
		if err := adaptor.HTTPHandler(check)(c); err != nil {
			return err
		}
		if data == nil {
			return nil // goauth already wrote the response
		}
		c.Locals("auth", *data)
		return c.Next()
	}
}
```

## Try it

Start the app (it creates `app.db`), then add a demo user in another terminal. The hash below is `goauth.HashPassword("a-long-passphrase")`:

```bash
go run .
```

```bash
sqlite3 app.db "INSERT INTO users (id, email, password_hash, roles) VALUES (
  '00000000-0000-0000-0000-000000000001', 'jane@example.com',
  '\$pbkdf2-sha256\$600000\$szJnPJf8L3F8EnWg2UwIZw\$sJsfkxgYgHKEUnDLV9gSN2eKkH9JBzV-1RINE7eTGiU',
  '[\"admin:rw\"]');"
```

Then exercise every route:

```bash
# Sign in and keep the session cookie.
curl -c jar.txt -X POST localhost:3000/api/login \
  -d '{"email":"jane@example.com","password":"a-long-passphrase"}'

# Call the API with the access token from the response.
curl -H "Authorization: Bearer $ACCESS_TOKEN" localhost:3000/api/me

# Refresh the access token.
curl -X POST localhost:3000/api/token/refresh -H "cft: 1" -H "Authorization: Bearer $REFRESH_TOKEN"

# Open the cookie-protected page.
curl -b jar.txt localhost:3000/dashboard
```

Browser sign-in: open [http://localhost:3000/dashboard](http://localhost:3000/dashboard) and click **Sign in with Google** (set `GOOGLE_CLIENT_ID` and `GOOGLE_CLIENT_SECRET` first).

## Keeping a handler in net/http

If you prefer, wrap a whole `net/http` chain instead of using `withAuth`:

```go
profile := auth.UseAPIAuthorization()(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
	me, _ := auth.GetAuthDataAPI(r) // works: same request
	json.NewEncoder(w).Encode(me)
}))
app.Get("/api/profile", adaptor.HTTPHandler(profile))
```

## Gate-only middleware

When the handler doesn't need to know the caller, `adaptor.HTTPMiddleware` is enough:

```go
app.Get("/api/ping", adaptor.HTTPMiddleware(auth.UseAPIAuthorization()), func(c fiber.Ctx) error {
	return c.SendString("pong")
})
```
